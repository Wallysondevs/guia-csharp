9. Abas: vários .cs (um projeto)

var conta = new Conta("Ana");
Console.WriteLine($"Saldo: {conta.Saldo:C}");

class Conta(string dono)
{
    public string Dono { get; } = dono;
    public decimal Saldo { get; private set; }
    public void Depositar(decimal valor) => Saldo += valor;
}

var conta = new Conta("Ana");
conta.Depositar(150m);
Console.WriteLine($"Saldo: {conta.Saldo:C}");
